\newcommand{\timeresplot}[7]{%
% Arguments:
% #1: benchmark instance
% #2: comma separated list of basecfg/style items
% #3: comma separated list of readable config names or 'false' if no legend should be shown
% #4: mintime
% #5: maxtime
% #6: minvalue
% #7: maxvalue
\ifthenelse{\equal{#6}{}}%
  {\pgfplotsset{currentvaluerange/.style={}}}%
  {\pgfplotsset{currentvaluerange/.style={ymin=#6,ymax=#7}}}%
\begin{tikzpicture}
\begin{axis}[
defaultplot,
xmin=#4,
xmax=#5,
currentvaluerange,
xmode=log,
xlabel=time (seconds),
title={\texttt{\detokenize{#1}}},
title style={font=\tiny,yshift=6pt},
%ymajorgrids,
%ytick= {1, 6, 60, 600, 1200, 1800 },
% yticklabels={$\le$1, 6, 60, 600, 1200, 1800},
ylabel=value bounds,
%ylabel={$ \mathsf{Pr}_{\max}^{\pomdp}\left(\costEventually{\costboundset} \ T\right)$},
%	y label style={at={(axis description cs:-0.1,0.5)},anchor=south},
log ticks with fixed point, % enable to avoid 10^-x notation
scaled x ticks=false,
%legend pos=north west,
legend style={nodes={scale=0.65, transform shape},inner sep=1pt, anchor=south, yshift=4pt, \plotlegendshift},
	]
\foreach \basecfg/\plotstyle in {#2}{%
		\edef\loopbody{
			\noexpand\addplot[\plotstyle, mark repeat=2, mark phase=2]
			table [x={\basecfg.#1.time},y={\basecfg.#1.result}, col sep=tab]
			{time_result.csv};
		}
		\loopbody
}
\ifthenelse{\equal{#3}{false}}{}{\legend{#3}}
\end{axis}
\ifthenelse{\equal{#3}{false}}{\boundingboxnolegend}{\boundingboxlegend}
\end{tikzpicture}%
}
